{{/* search-sections — Divide el HTML renderizado de una página en secciones H2/H3 para la búsqueda. Devuelve [{h: encabezado, a: ancla, l: nivel, x: texto plano}] (la primera, sin encabezado, es la intro). Se descartan bloques de código, scripts y estilos; el texto de cada sección se trunca a site.Params.searchSectionChars (por defecto 1200) para no inflar el índice. */}} {{ $max := site.Params.searchSectionChars | default 1200 }} {{ $html := .Content }} {{ $html = replaceRE `(?s)]*>.*?` " " $html }} {{ $html = replaceRE `(?s)]*>.*?` " " $html }} {{ $html = replaceRE `(?s)]*>.*?` " " $html }} {{ $html = replaceRE `(?s)
.*?
` " " $html }} {{ $html = replaceRE `]*>[^<]*` " " $html }} {{/* Solo los encabezados del render hook (llevan .doc-anchor); los de las demos se quedan como texto. */}} {{ $html = replaceRE `(?s)]*>(.*?)]*>#` "\x1eSEC$1\x1f$2\x1f$3\x1eEND" $html }} {{/* Espacio antes de cada etiqueta para que las celdas y bloques no se peguen al quitar el HTML. */}} {{ $html = replaceRE `<` " <" $html }} {{ $out := slice }} {{ range $i, $part := split $html "\x1eSEC" }} {{ if eq $i 0 }} {{ $x := $part | plainify | htmlUnescape | replaceRE `\s+` " " | replaceRE ` ([,.;:!?)\]])` "$1" | strings.TrimSpace }} {{ if $x }}{{ $out = $out | append (dict "h" "" "a" "" "l" 1 "x" (partial "search-clip.html" (dict "text" $x "max" $max))) }}{{ end }} {{ else }} {{ $headBody := split $part "\x1eEND" }} {{ $meta := split (index $headBody 0) "\x1f" }} {{ if ge (len $meta) 3 }} {{ $h := index $meta 2 | plainify | htmlUnescape | replaceRE `\s+` " " | replaceRE ` ([,.;:!?)\]])` "$1" | strings.TrimSpace }} {{ $x := "" }} {{ if gt (len $headBody) 1 }}{{ $x = index $headBody 1 | plainify | htmlUnescape | replaceRE `\s+` " " | replaceRE ` ([,.;:!?)\]])` "$1" | strings.TrimSpace }}{{ end }} {{ $out = $out | append (dict "h" $h "a" (index $meta 1) "l" (int (index $meta 0)) "x" (partial "search-clip.html" (dict "text" $x "max" $max))) }} {{ end }} {{ end }} {{ end }} {{ return $out }}
